import { randomUUID } from "node:crypto";
import type { Db } from "../db.js";
import type { Booking, BookingView, SessionWithCapacity } from "../types.js";
import {
  countConfirmedBookings,
  hasCapacity,
  remainingCapacity,
} from "./capacity.js";

export class BookingError extends Error {
  constructor(
    message: string,
    public readonly code:
      | "NOT_FOUND"
      | "FULL"
      | "DUPLICATE"
      | "INVALID"
      | "ALREADY_CANCELLED"
  ) {
    super(message);
    this.name = "BookingError";
  }
}

function mapBooking(row: Record<string, unknown>): Booking {
  return {
    id: String(row.id),
    sessionId: String(row.session_id),
    userId: String(row.user_id),
    status: row.status as Booking["status"],
    createdAt: String(row.created_at),
  };
}

export function listSessionsForEvent(
  db: Db,
  eventId: string
): SessionWithCapacity[] {
  const rows = db
    .prepare(
      `SELECT id, event_id, starts_at, ends_at, capacity
       FROM sessions
       WHERE event_id = ?
       ORDER BY starts_at ASC`
    )
    .all(eventId) as Array<Record<string, unknown>>;

  return rows.map((row) => {
    const id = String(row.id);
    const capacity = Number(row.capacity);
    const bookedCount = countConfirmedBookings(db, id);
    return {
      id,
      eventId: String(row.event_id),
      startsAt: String(row.starts_at),
      endsAt: String(row.ends_at),
      capacity,
      bookedCount,
      remainingCapacity: capacity - bookedCount,
    };
  });
}

export function getSessionWithCapacity(
  db: Db,
  sessionId: string
): SessionWithCapacity | null {
  const row = db
    .prepare(
      `SELECT id, event_id, starts_at, ends_at, capacity
       FROM sessions WHERE id = ?`
    )
    .get(sessionId) as Record<string, unknown> | undefined;
  if (!row) return null;
  const capacity = Number(row.capacity);
  const bookedCount = countConfirmedBookings(db, sessionId);
  return {
    id: String(row.id),
    eventId: String(row.event_id),
    startsAt: String(row.starts_at),
    endsAt: String(row.ends_at),
    capacity,
    bookedCount,
    remainingCapacity: capacity - bookedCount,
  };
}

/**
 * Create an individual (1-seat) confirmed booking.
 *
 * TODO(challenge): accept partySize / attendeeIds for group bookings.
 * TODO(challenge): if insufficient capacity, offer waitlist instead of FULL.
 */
export function createBooking(
  db: Db,
  sessionId: string,
  userId: string
): Booking {
  const session = db
    .prepare(`SELECT id FROM sessions WHERE id = ?`)
    .get(sessionId);
  if (!session) {
    throw new BookingError("Session not found", "NOT_FOUND");
  }

  const user = db.prepare(`SELECT id FROM users WHERE id = ?`).get(userId);
  if (!user) {
    throw new BookingError("User not found", "NOT_FOUND");
  }

  const existing = db
    .prepare(
      `SELECT id FROM bookings
       WHERE session_id = ? AND user_id = ? AND status = 'confirmed'`
    )
    .get(sessionId, userId);
  if (existing) {
    throw new BookingError(
      "User already has a confirmed booking for this session",
      "DUPLICATE"
    );
  }

  if (!hasCapacity(db, sessionId, 1)) {
    // TODO(challenge): join waitlist when full / partial group fit fails.
    throw new BookingError("Session is fully booked", "FULL");
  }

  const booking: Booking = {
    id: randomUUID(),
    sessionId,
    userId,
    status: "confirmed",
    createdAt: new Date().toISOString(),
  };

  db.prepare(
    `INSERT INTO bookings (id, session_id, user_id, status, created_at)
     VALUES (?, ?, ?, ?, ?)`
  ).run(
    booking.id,
    booking.sessionId,
    booking.userId,
    booking.status,
    booking.createdAt
  );

  return booking;
}

/**
 * Cancel a confirmed booking and free one place.
 *
 * TODO(challenge): after cancel, evaluate waitlist and auto-promote
 * (fairness / fit rules are a stakeholder CLARIFY topic — do not hard-code answers).
 */
export function cancelBooking(db: Db, bookingId: string): Booking {
  const row = db
    .prepare(
      `SELECT id, session_id, user_id, status, created_at
       FROM bookings WHERE id = ?`
    )
    .get(bookingId) as Record<string, unknown> | undefined;

  if (!row) {
    throw new BookingError("Booking not found", "NOT_FOUND");
  }

  if (row.status === "cancelled") {
    throw new BookingError("Booking is already cancelled", "ALREADY_CANCELLED");
  }

  db.prepare(`UPDATE bookings SET status = 'cancelled' WHERE id = ?`).run(
    bookingId
  );

  // Capacity is freed by status change (confirmed count drops).
  // TODO(challenge): trigger waitlist promotion here.

  return {
    ...mapBooking(row),
    status: "cancelled",
  };
}

export function listUserBookings(db: Db, userId: string): BookingView[] {
  const rows = db
    .prepare(
      `SELECT
         b.id, b.session_id, b.user_id, b.status, b.created_at,
         e.title AS event_title,
         s.starts_at AS session_starts_at,
         s.ends_at AS session_ends_at,
         e.venue,
         u.name AS user_name
       FROM bookings b
       JOIN sessions s ON s.id = b.session_id
       JOIN events e ON e.id = s.event_id
       JOIN users u ON u.id = b.user_id
       WHERE b.user_id = ?
       ORDER BY s.starts_at ASC, b.created_at ASC`
    )
    .all(userId) as Array<Record<string, unknown>>;

  return rows.map((row) => ({
    id: String(row.id),
    sessionId: String(row.session_id),
    userId: String(row.user_id),
    status: row.status as Booking["status"],
    createdAt: String(row.created_at),
    eventTitle: String(row.event_title),
    sessionStartsAt: String(row.session_starts_at),
    sessionEndsAt: String(row.session_ends_at),
    venue: String(row.venue),
    userName: String(row.user_name),
  }));
}

export function assertRemaining(
  db: Db,
  sessionId: string,
  expected: number
): void {
  const actual = remainingCapacity(db, sessionId);
  if (actual !== expected) {
    throw new Error(
      `Capacity mismatch for ${sessionId}: expected ${expected}, got ${actual}`
    );
  }
}

import type { Db } from "../db.js";

/**
 * Confirmed bookings consume one place each (individual booking model).
 * remaining = capacity - confirmedCount
 *
 * TODO(challenge): when group bookings exist, capacity must account for party size.
 * TODO(challenge): waitlist entries must not consume capacity until promoted.
 */
export function countConfirmedBookings(db: Db, sessionId: string): number {
  const row = db
    .prepare(
      `SELECT COUNT(*) AS count
       FROM bookings
       WHERE session_id = ? AND status = 'confirmed'`
    )
    .get(sessionId) as { count: number } | undefined;
  return Number(row?.count ?? 0);
}

export function getSessionCapacity(db: Db, sessionId: string): number | null {
  const row = db
    .prepare(`SELECT capacity FROM sessions WHERE id = ?`)
    .get(sessionId) as { capacity: number } | undefined;
  return row ? Number(row.capacity) : null;
}

export function remainingCapacity(db: Db, sessionId: string): number | null {
  const capacity = getSessionCapacity(db, sessionId);
  if (capacity === null) return null;
  return capacity - countConfirmedBookings(db, sessionId);
}

export function hasCapacity(db: Db, sessionId: string, seats = 1): boolean {
  const remaining = remainingCapacity(db, sessionId);
  if (remaining === null) return false;
  return remaining >= seats;
}

import cors from "cors";
import express, { type Express, type Request, type Response } from "express";
import type { Db } from "./db.js";
import {
  BookingError,
  cancelBooking,
  createBooking,
  getSessionWithCapacity,
  listSessionsForEvent,
  listUserBookings,
} from "./services/bookingService.js";
import type { Event, EventDetail, User } from "./types.js";

export function createApp(db: Db): Express {
  const app = express();
  app.use(cors());
  app.use(express.json());

  app.get("/api/health", (_req, res) => {
    res.json({ ok: true, service: "eventhub-api" });
  });

  app.get("/api/users", (_req, res) => {
    const rows = db
      .prepare(`SELECT id, name, email FROM users ORDER BY name ASC`)
      .all() as Array<Record<string, unknown>>;
    const users: User[] = rows.map((r) => ({
      id: String(r.id),
      name: String(r.name),
      email: String(r.email),
    }));
    res.json(users);
  });

  app.get("/api/events", (_req, res) => {
    const rows = db
      .prepare(
        `SELECT id, title, description, venue FROM events ORDER BY title ASC`
      )
      .all() as Array<Record<string, unknown>>;
    const events: Event[] = rows.map((r) => ({
      id: String(r.id),
      title: String(r.title),
      description: String(r.description),
      venue: String(r.venue),
    }));
    res.json(events);
  });

  app.get("/api/events/:id", (req, res) => {
    const row = db
      .prepare(
        `SELECT id, title, description, venue FROM events WHERE id = ?`
      )
      .get(req.params.id) as Record<string, unknown> | undefined;
    if (!row) {
      res.status(404).json({ error: "Event not found" });
      return;
    }
    const detail: EventDetail = {
      id: String(row.id),
      title: String(row.title),
      description: String(row.description),
      venue: String(row.venue),
      sessions: listSessionsForEvent(db, String(row.id)),
    };
    res.json(detail);
  });

  app.get("/api/sessions/:id", (req, res) => {
    const session = getSessionWithCapacity(db, req.params.id);
    if (!session) {
      res.status(404).json({ error: "Session not found" });
      return;
    }
    res.json(session);
  });

  app.get("/api/users/:id/bookings", (req, res) => {
    const user = db
      .prepare(`SELECT id FROM users WHERE id = ?`)
      .get(req.params.id);
    if (!user) {
      res.status(404).json({ error: "User not found" });
      return;
    }
    res.json(listUserBookings(db, req.params.id));
  });

  app.post("/api/bookings", (req: Request, res: Response) => {
    const { sessionId, userId } = req.body ?? {};
    if (typeof sessionId !== "string" || typeof userId !== "string") {
      res.status(400).json({
        error: "sessionId and userId are required strings",
      });
      return;
    }
    // Individual booking only — one seat.
    // TODO(challenge): accept partySize / attendees for group bookings.
    // TODO(challenge): waitlist when capacity insufficient.
    try {
      const booking = createBooking(db, sessionId, userId);
      res.status(201).json(booking);
    } catch (err) {
      respondBookingError(res, err);
    }
  });

  app.delete("/api/bookings/:id", (req, res) => {
    try {
      const booking = cancelBooking(db, req.params.id);
      res.json(booking);
    } catch (err) {
      respondBookingError(res, err);
    }
  });

  return app;
}

function respondBookingError(res: Response, err: unknown): void {
  if (err instanceof BookingError) {
    const status =
      err.code === "NOT_FOUND"
        ? 404
        : err.code === "FULL" ||
            err.code === "DUPLICATE" ||
            err.code === "ALREADY_CANCELLED" ||
            err.code === "INVALID"
          ? 409
          : 400;
    res.status(status).json({ error: err.message, code: err.code });
    return;
  }
  console.error(err);
  res.status(500).json({ error: "Internal server error" });
}

import { DatabaseSync } from "node:sqlite";
import path from "node:path";
import { fileURLToPath } from "node:url";

const __dirname = path.dirname(fileURLToPath(import.meta.url));

export type Db = DatabaseSync;

export function defaultDbPath(): string {
  return path.resolve(__dirname, "../../../data/eventhub.db");
}

export function openDatabase(dbPath: string = defaultDbPath()): Db {
  const db = new DatabaseSync(dbPath);
  db.exec("PRAGMA foreign_keys = ON;");
  migrate(db);
  return db;
}

export function openMemoryDatabase(): Db {
  const db = new DatabaseSync(":memory:");
  db.exec("PRAGMA foreign_keys = ON;");
  migrate(db);
  return db;
}

export function migrate(db: Db): void {
  db.exec(`
    CREATE TABLE IF NOT EXISTS users (
      id TEXT PRIMARY KEY,
      name TEXT NOT NULL,
      email TEXT NOT NULL UNIQUE
    );

    CREATE TABLE IF NOT EXISTS events (
      id TEXT PRIMARY KEY,
      title TEXT NOT NULL,
      description TEXT NOT NULL,
      venue TEXT NOT NULL
    );

    CREATE TABLE IF NOT EXISTS sessions (
      id TEXT PRIMARY KEY,
      event_id TEXT NOT NULL REFERENCES events(id) ON DELETE CASCADE,
      starts_at TEXT NOT NULL,
      ends_at TEXT NOT NULL,
      capacity INTEGER NOT NULL CHECK (capacity >= 0)
    );

    -- Individual booking only (one seat per booking row).
    -- TODO(challenge): group bookings — party size / attendee list; keep groups together.
    -- TODO(challenge): waitlists — queue when capacity insufficient; auto-promote on cancel.
    CREATE TABLE IF NOT EXISTS bookings (
      id TEXT PRIMARY KEY,
      session_id TEXT NOT NULL REFERENCES sessions(id) ON DELETE CASCADE,
      user_id TEXT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
      status TEXT NOT NULL CHECK (status IN ('confirmed', 'cancelled')),
      created_at TEXT NOT NULL
    );

    CREATE INDEX IF NOT EXISTS idx_bookings_session_status
      ON bookings(session_id, status);

    CREATE INDEX IF NOT EXISTS idx_bookings_user
      ON bookings(user_id);

    CREATE INDEX IF NOT EXISTS idx_sessions_event
      ON sessions(event_id);
  `);
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(path, {
    headers: { "Content-Type": "application/json", ...(init?.headers ?? {}) },
    ...init,
  });
  const body = await res.json().catch(() => ({}));
  if (!res.ok) {
    const message =
      typeof body.error === "string" ? body.error : `Request failed (${res.status})`;
    throw new Error(message);
  }
  return body as T;
}

export const api = {
  listUsers: () => request<User[]>("/api/users"),
  listEvents: () => request<EventSummary[]>("/api/events"),
  getEvent: (id: string) => request<EventDetail>(`/api/events/${id}`),
  listBookings: (userId: string) =>
    request<BookingView[]>(`/api/users/${userId}/bookings`),
  createBooking: (sessionId: string, userId: string) =>
    request<{ id: string }>("/api/bookings", {
      method: "POST",
      body: JSON.stringify({ sessionId, userId }),
    }),
  cancelBooking: (bookingId: string) =>
    request<{ id: string; status: string }>(`/api/bookings/${bookingId}`, {
      method: "DELETE",
    }),
};

import { useEffect, useState } from "react";
import { Link, Navigate, Route, Routes } from "react-router-dom";
import { api } from "./api";
import { EventDetailPage } from "./pages/EventDetailPage";
import { EventsPage } from "./pages/EventsPage";
import { MyBookingsPage } from "./pages/MyBookingsPage";
import type { User } from "./types";

const USER_KEY = "eventhub.currentUserId";

export function App() {
  const [users, setUsers] = useState<User[]>([]);
  const [userId, setUserId] = useState(
    () => localStorage.getItem(USER_KEY) ?? ""
  );
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    api
      .listUsers()
      .then((list) => {
        setUsers(list);
        setUserId((current) => {
          if (current && list.some((u) => u.id === current)) return current;
          const first = list[0]?.id ?? "";
          if (first) localStorage.setItem(USER_KEY, first);
          return first;
        });
      })
      .catch((err: Error) => setError(err.message));
  }, []);

  function onUserChange(id: string) {
    setUserId(id);
    localStorage.setItem(USER_KEY, id);
  }

  const currentUser = users.find((u) => u.id === userId) ?? null;

  return (
    <div className="shell">
      <header className="topbar">
        <div className="brand-block">
          <Link to="/" className="brand">
            EventHub
          </Link>
          <p className="tagline">Book a place on an event session</p>
        </div>
        <nav className="nav">
          <Link to="/">Events</Link>
          <Link to="/bookings">My bookings</Link>
        </nav>
        <label className="user-switcher">
          <span>Acting as</span>
          <select
            value={userId}
            onChange={(e) => onUserChange(e.target.value)}
            disabled={users.length === 0}
          >
            {users.map((u) => (
              <option key={u.id} value={u.id}>
                {u.name}
              </option>
            ))}
          </select>
        </label>
      </header>

      {error && <p className="banner error">{error}</p>}

      <main className="main">
        <Routes>
          <Route path="/" element={<EventsPage />} />
          <Route
            path="/events/:id"
            element={
              currentUser ? (
                <EventDetailPage user={currentUser} />
              ) : (
                <p className="muted">Loading users…</p>
              )
            }
          />
          <Route
            path="/bookings"
            element={
              currentUser ? (
                <MyBookingsPage user={currentUser} />
              ) : (
                <p className="muted">Loading users…</p>
              )
            }
          />
          <Route path="*" element={<Navigate to="/" replace />} />
        </Routes>
      </main>

      <footer className="footer">
        <p>
          Individual bookings only.{" "}
          {/* TODO(challenge): group bookings + waitlists — intentionally not built. */}
          Group bookings and waitlists are out of scope for this starter.
        </p>
      </footer>
    </div>
  );
}

# Facilitator notes — Day 4 EventHub starter

For trainers only. Delegates should use `README.md` and `USER_GUIDE.md`, plus the challenge brief you hand out.

## Alignment sources

- Challenge product brief: `/cursor/stores/self/docs/day4-eventhub-brief.md`
- Micro alignment (skills / agents / exceptions / human control): `/cursor/stores/self/docs/day4-micro-alignment.md`
- Elite agents PDF: `/cursor/stores/self/docs/QABAIDSGC-M-Build-Elite-AI-Agent-Teams.pdf`

## Day 4 flow (summary)

1. **Understand** — Repository Discovery Report on this starter  
2. **Specify** — Feature Spec including unanswered questions  
3. **Plan** — UI / API / domain / persistence / tests breakdown  
4. **Skills** — teams choose reusable skills (assess what and why)  
5. **Agent team** — justified roles, boundaries, human-owned decisions  
6. **Implement** — group bookings + waitlists; keep individual booking green  
7. **Trainer event cards** — mid-day facilitation prompts (below)  
8. **Independent validation** — validator must not silently fix production code  
9. **Human release decision**

Micro learning journey (design team → implement roles → connect handoffs → exceptions → retain human control) follows QABAIDSGC-M; Day 4 applies it to **EventHub**, not remediation or observability labs.

## What this starter provides

- Working individual booking (browse, capacity, book, cancel, my bookings)
- React → Node → SQLite; `npm install` && `npm run dev`
- Seeded events/sessions with varied capacity and existing bookings
- 40 automated tests as regression bait
- Explicit TODOs for groups/waitlists — **no** partial implementations

## Ambiguities — do not pre-answer

Hold stakeholder answers until teams **CLARIFY**:

| Topic | Do not volunteer |
| --- | --- |
| Partial availability (e.g. 20 capacity, 17 booked, request 4) | Groups stay together / never partially allocate — reveal only when asked |
| Waitlist fairness (3 free; queue 4, 1, 5) | Order vs first-that-fits — reveal only when asked |
| Explainability of waitlist allocation | Teams must design how stakeholders see *why* |

Point teams at the brief; resist solving the product for them.

## Mid-day event cards (facilitation prompts only)

These are **trainer interventions**, not in-app features. Use verbally or as printed cards.

1. **Cancellation skip behaviour** — A cancellation frees seats that could promote someone from the waitlist. Prompt: who gets promoted, and what evidence shows why someone was skipped?  
2. **Race overbooking** — Concurrent booking requests near capacity. Prompt: can the system over-allocate? What test proves it cannot?  
3. **Agent scope creep** — An agent starts “helpfully” implementing unrelated UI polish or changing individual-booking rules. Prompt: which exception route (CLARIFY / ESCALATE / REVISE / STOP) and who owns the decision?

Do not ship these as product toggles in the starter.

- [ ] `npm install` && `npm test` pass on a clean machine (Node 22+)
- [ ] `npm run dev` serves UI + API; seed data visible
- [ ] Challenge brief and assessment criteria ready for handout
- [ ] Skills/agents are **not** pre-seeded with the solution
- [ ] Plan at least one REVISE and one ESCALATE/STOP exercise
- [ ] Human decision artefact is typed by a named person

## Handing to delegates

Zip or clone `docs/day4-starter/`. Tell them: product works; extend per brief; existing tests are the contract for individual booking; ask stakeholders when rules are unclear.

# EventHub (Day 4 starter)

Working **individual** event booking app for YouView / QA training.

Delegates inherit this product and extend it with **group bookings** and **waitlists**. Those features are intentionally absent — leave the TODOs alone until the challenge brief says otherwise.

## What works today

- Browse events and sessions
- See remaining capacity
- Book an individual place
- Cancel a booking
- View your bookings

## Stack

| Layer | Tech |
| --- | --- |
| UI | React + TypeScript (Vite) |
| API | Node + TypeScript (Express) |
| DB | SQLite via Node built-in `node:sqlite` |

No Docker. No external services. No API keys.

## Quick start

Requires **Node.js 22+** (for `node:sqlite`).

```bash
npm install
npm run dev
```

- UI: http://localhost:5173  
- API: http://localhost:4000  

On first API start, SQLite is created and seeded at `data/eventhub.db`.

Re-seed (wipes local DB):

```bash
npm run seed
```

## Tests

```bash
npm test
```

Backend Vitest suite (**40 tests**) covers capacity, booking, cancel, duplicates, overbooking, and seed integrity. Treat failures as regression bait when adding groups/waitlists.

## Domain model

```
User ──< Booking >── Session >── Event
              │
         status: confirmed | cancelled
```

- **Event** — titled offering with venue
- **Session** — timed occurrence with `capacity`
- **User** — attendee (demo switcher in the UI; no auth)
- **Booking** — one confirmed seat per user per session
- **Remaining capacity** = `session.capacity − count(confirmed bookings)`

Cancelled bookings do not consume capacity.

## Intentionally missing

| Capability | Starter stance |
| --- | --- |
| Group bookings | TODO comments only — no party size / partial allocation |
| Waitlists | TODO comments only — no queue or auto-promotion |
| Waitlist explainability | Out of scope until teams design it |

Do **not** add partial implementations that spoil the challenge.

## Project layout

```
backend/     Express API, SQLite, seed, Vitest
frontend/    React UI
data/        SQLite file (generated)
```

- `USER_GUIDE.md` — delegate walkthrough
- `FACILITATOR.md` — trainer notes (do not hand to delegates as the brief)

# EventHub — user guide (delegates)

## Run the app

```bash
npm install
npm run dev
```

Open http://localhost:5173. The API runs on port 4000; the Vite dev server proxies `/api`.

## Demo users

Use **Acting as** in the header to switch seeded users (Alice, Ben, Cara, Diego, Elena, Farah). There is no login — this is a training demo.

## Browse and book

1. Open **Events**.
2. Choose an event → **View sessions**.
3. Check **places remaining**.
4. Click **Book place** (one seat for the current user).
5. Open **My bookings** to confirm.

Full sessions show **Full** and cannot be booked.

## Cancel

On **My bookings**, cancel a confirmed booking. Capacity on that session increases by one. Cancelled rows remain visible for history.

## Seeded scenarios worth noticing

| Session | Situation |
| --- | --- |
| Keynote AM | Several places already taken |
| Keynote PM | Empty — easy booking |
| Workshop A | Capacity 7, six booked — one seat left |
| Tiny panel slot | Fully booked |
| Workshop B | Open (one historical cancelled booking only) |

Re-seed after experiments:

```bash
npm run seed
```

## Tests before you change behaviour

```bash
npm test
```

Existing tests lock **individual** booking rules. Keep them green while you add group/waitlist behaviour (extend or add tests — do not delete coverage of current behaviour without reason).

## Challenge boundary

You will be asked to add **group bookings** and **waitlists**. The starter only supports individual bookings. Search the codebase for `TODO(challenge)` for hooks — do not invent stakeholder answers; use **CLARIFY** when rules are ambiguous.
